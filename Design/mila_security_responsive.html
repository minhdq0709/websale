auth.middleware.js — Xác thực JWT
const jwt = require('jsonwebtoken'); const redis = require('../config/redis'); module.exports = async (req, res, next) => { const header = req.headers.authorization; if (!header?.startsWith('Bearer ')) return res.status(401).json({ code:'NO_TOKEN', message:'Chưa đăng nhập.' }); const token = header.split(' ')[1]; // Kiểm tra blacklist (token bị revoke khi logout) const isBlacklisted = await redis.get(`bl:${token}`); if (isBlacklisted) return res.status(401).json({ code:'TOKEN_REVOKED', message:'Token đã bị thu hồi.' }); try { const decoded = jwt.verify(token, process.env.JWT_SECRET); req.user = { id: decoded.sub, phone: decoded.phone, role: decoded.role }; next(); } catch (err) { const code = err.name === 'TokenExpiredError' ? 'TOKEN_EXPIRED' : 'INVALID_TOKEN'; res.status(401).json({ code, message: 'Token không hợp lệ hoặc hết hạn.' }); } };
req.user được gán401 nếu sai/hết hạn/bị revoke

role.middleware.js — Kiểm tra quyền RBAC
// Dùng sau auth.middleware // requireRole('admin') — chỉ admin // requireAny('admin','staff') — admin hoặc staff exports.requireRole = (...roles) => (req, res, next) => { if (!roles.includes(req.user?.role)) return res.status(403).json({ code: 'FORBIDDEN', message: `Yêu cầu role: ${roles.join(' hoặc ')}.` }); next(); }; exports.requireOwner = (paramKey = 'userId') => (req, res, next) => { const isAdmin = req.user.role === 'admin'; const isOwner = req.user.id === parseInt(req.params[paramKey]); if (!isAdmin && !isOwner) return res.status(403).json({ code: 'NOT_OWNER', message: 'Không có quyền.' }); next(); };
403 nếu sai rolenext() nếu đúng

validate.middleware.js — AJV Validation + Sanitize
const Ajv = require('ajv'); const addFormats = require('ajv-formats'); const createDOMPurify = require('dompurify'); const { JSDOM } = require('jsdom'); const ajv = new Ajv({ allErrors: true, coerceTypes: true }); addFormats(ajv); const purify = createDOMPurify(new JSDOM('').window); function sanitize(obj) { if (typeof obj === 'string') return purify.sanitize(obj.trim()); if (typeof obj === 'object' && obj) return Object.fromEntries(Object.entries(obj).map(([k,v]) => [k, sanitize(v)])); return obj; } module.exports = (schema) => { const validate = ajv.compile(schema); return (req, res, next) => { req.body = sanitize(req.body); // strip HTML/script injection if (!validate({ body: req.body, params: req.params, query: req.query })) return res.status(400).json({ code:'VALIDATION_ERROR', errors: validate.errors }); next(); }; };

Cách gắn role vào route
const { requireRole, requireOwner } = require('../middleware/role.middleware'); const auth = require('../middleware/auth.middleware'); // Public — không cần login router.get('/products', ctrl.list); // Customer trở lên — cần login, mọi role đều đi qua router.post('/orders', auth, validate(schema), ctrl.create); // Chỉ xem đơn hàng của chính mình (customer) HOẶC admin/staff router.get('/orders/:id', auth, requireOwner('userId'), ctrl.getOne); // Chỉ staff hoặc admin router.put('/orders/:id/status', auth, requireRole('admin','staff'), ctrl.updateStatus); // Chỉ admin router.get('/admin/reports', auth, requireRole('admin'), ctrl.report); router.delete('/users/:id', auth, requireRole('admin'), ctrl.deleteUser);
Dùng 2 token: Access Token ngắn hạn (15 phút) + Refresh Token dài hạn (7 ngày, lưu httpOnly cookie). Rotation mỗi lần refresh.
⚡ Access Token (JWT)
Lưu ở sessionStorage FE. Gửi trong header Authorization. Hết hạn sau 15 phút.
// Header { "alg": "HS256", "typ": "JWT" } // Payload { "sub": 42, // user_id "phone": "0912...", "role": "customer", // guest|customer|staff|admin "iat": 1747382400, // issued at "exp": 1747383300 // +15 phút } // Ký bằng JWT_SECRET (64+ ký tự random)
🔄 Refresh Token
Lưu ở httpOnly cookie (JS không đọc được). Hết hạn 7 ngày. Rotation: mỗi lần dùng là tạo cái mới và revoke cái cũ.
// DB table: refresh_tokens { id, user_id, token_hash, // SHA-256(random 64 bytes) device_info, // User-Agent ip_address, created_at, expires_at, revoked_at // null nếu còn hiệu lực } // Cookie options { httpOnly: true, secure: true, sameSite: 'Lax', maxAge: 7*24*3600*1000 }
Luồng Token Lifecycle
Login: POST /auth/login → { accessToken } + Set-Cookie: refreshToken=... Silent Refresh (FE tự động khi access token hết hạn): POST /auth/refresh (gửi cookie tự động) → { accessToken mới } → Server revoke refresh token cũ, tạo refresh token mới (rotation) Logout: POST /auth/logout → revoke refresh token ở DB → Thêm access token vào Redis blacklist (TTL = còn lại của token) → Clear cookie Phát hiện tấn công: Nếu refresh token đã bị revoke mà vẫn được dùng lại → Revoke TẤT CẢ session của user đó (full logout) → Gửi cảnh báo email cho user
OTP Token (cho đăng ký & thanh toán QR)
// DB table: otp_codes { phone, email, otp_hash, // SHA-256(6-digit OTP) — không lưu OTP thô purpose, // 'register' | 'payment' | 'forgot_password' attempts, // max 3 lần sai → lock expires_at // +5 phút } // verifyToken sau khi OTP đúng (JWT 10 phút, 1 lần dùng) { phone, purpose, verified: true, exp: now+600 } // Truyền vào /auth/register hoặc /orders/create
Toàn bộ biện pháp bảo mật theo OWASP Top 10 — từ transport layer đến application layer.

💉 SQL Injection Prevention
Dùng Prepared Statements (mysql2). KHÔNG BAO GIỜ nối chuỗi vào query. Validate input trước bằng AJV.
// ✅ Đúng — Prepared Statement db.query( 'SELECT * FROM products WHERE id = ?', [req.params.id] // ← tham số hóa ); // ❌ Sai — SQL Injection db.query('SELECT * FROM products WHERE id = ' + req.params.id); // nguy hiểm!
🧹 XSS Prevention
Sanitize tất cả input qua DOMPurify trước khi lưu DB. Content-Security-Policy header chặn inline script. Output encode khi render HTML.
// Middleware sanitize mọi string input req.body = sanitize(req.body); // purify.sanitize(str) strips all <script> // và event handlers (onerror, onclick...) // CSP Header (Nginx) Content-Security-Policy: default-src 'self'; script-src 'self' cdn.jsdelivr.net; img-src 'self' images.unsplash.com;
🔐 Password Security
bcrypt với 12 rounds. Không lưu plaintext. Không trả password về bất kỳ API nào. Độ phức tạp tối thiểu 6 ký tự.
const bcrypt = require('bcrypt'); // Hash khi tạo tài khoản const hash = await bcrypt.hash(password, 12); // Verify khi login (constant-time compare) const ok = await bcrypt.compare(input, hash); // Không bao giờ select password ra ngoài SELECT id, phone, email, role FROM users // ← không có cột 'password'
🛡 CSRF Protection
SameSite=Lax trên cookie. Double-Submit pattern cho form mutation. Origin header check ở middleware.
// Middleware kiểm tra Origin if (req.method !== 'GET') { const origin = req.headers.origin; const allowed = process.env.CLIENT_URL; if (origin !== allowed) return res.status(403).json({ code: 'CSRF_BLOCKED' }); } // Cookie SameSite=Lax tự chặn // cross-site form submission
